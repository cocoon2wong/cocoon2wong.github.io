import paperRepos from "/src/data/repo_data_paper.json";

## Functional Repos

Here are additional code repos for training our trajectory prediction models, including the training structure, dataset processings codes, and trajectory visualization codes.

<div class="repo-container">
  {funcRepos.map((repo) => {
    const pillItems = [
      { label: "View", href: `https://github.com/${repo.user}/${repo.repo_name}` },
      repo.homepage ? { label: "Homepage", href: repo.homepage } : null,
    ].filter(Boolean);

    return (
      <LiquidGlass hoverable class="repo-box" themeColor="#ffffff">
        <div class="repo-margin-container">
          <div class="repo-vbox">
            <div class="repo-info">
              <div class="repo-title">{repo.display_name}</div>
              <p class="repo-desc">{repo.desc}</p>
            </div>
            <div class="repo-bottom">
              <Pill items={pillItems} />
            </div>
          </div>
        </div>
      </LiquidGlass>
    );
  })}
</div>
